# Useful commands

## Build an image from the Dockerfile in the current directory
docker build -t my-image-name .

## Run a container (interactive, remove when done)
docker run --rm -it my-image-name

## Run in the background (detached)
docker run -d --name my-container my-image-name

## See running containers
docker ps

## See logs from a running container
docker logs my-container

## Stop / remove a container
docker stop my-container
docker rm my-container

## Shell into a running container (great for debugging)
docker exec -it my-container bash

# Dockerfile

## Start FROM a base image (someone else's pre-built environment)
FROM node:20-slim

## Set a working directory inside the container
WORKDIR /app

## Copy your dependency manifest first (for layer caching)
COPY package*.json ./

## Install dependencies
RUN npm ci

## Copy the rest of your source code
COPY . .

## Build the project
RUN npm run build

## What command runs when the container starts
CMD ["node", "dist-electron/main.js"]
